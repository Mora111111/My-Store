       if (empty($words)) {
           return $this->getAll();
       }
       $conditions = [];
       $params = [];
       foreach ($words as $word) {
           $conditions[] = "(name LIKE ? OR email LIKE ?)";
           $params[] = '%' . $word . '%';
           $params[] = '%' . $word . '%';
       }
       $sql = "SELECT * FROM elogin WHERE " . implode(' AND ', $conditions) . " ORDER BY id DESC";
