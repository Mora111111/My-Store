ADMIN COMMENTS EMAIL DISPLAY FEATURE:

1. MODIFY app/Models/Comment.php

   public function getAllWithUserEmail(): array {
       $stmt = $this->db->query("
           SELECT c.*, p.title as product_title, p.image_url,
                  e.email as customer_email
           FROM product_comments c
           JOIN products p ON c.product_id = p.id
           LEFT JOIN elogin e ON c.customer_name = e.name
           ORDER BY c.created_at DESC
       ");
       return $stmt->fetchAll();
   }
   
   - ALSO ADD method to get user by name (for cases where name doesn't match exactly):
   public function getUserByName(string $name): array|false {

       $stmt->execute([$name]);
       return $stmt->fetch();
   }
